\documentclass[a4paper]{_extensions/modeloufcg/modeloUFCG}

\usepackage[portuguese]{babel}

\usepackage[utf8]{inputenc}
\usepackage{fancyhdr}

\usepackage{array}
\usepackage{calc}

%% Usa uma unica familia serifada em todo o relatorio. A classe carrega
%% Palatino como fonte romana; igualar \sfdefault a \rmdefault tambem cobre
%% elementos internos da classe que originalmente solicitavam Helvetica.
\renewcommand{\sfdefault}{\rmdefault}
%% Fixa cada figura exatamente onde e gerada ([H]). Sem isso, floats podem
%% "flutuar" livremente e, neste documento (muitas figuras grandes em
%% sequencia), acabam se deslocando para muito longe do texto que os
%% referencia -- inclusive interferindo na quebra de pagina de tabelas
%% longtable vizinhas.
\usepackage{float}
\floatplacement{figure}{H}

%% Evita que tabelas curtas comecem no fim da pagina e sejam interrompidas
%% por figuras na pagina seguinte. Tabelas realmente longas continuam
%% podendo quebrar normalmente entre paginas.
\BeforeBeginEnvironment{longtable}{\Needspace{12\baselineskip}}

%% A classe modeloUFCG.cls calcula largura/altura de texto e margens
%% assumindo o formato de trim de um periodico (174mm x 247mm), mas o PDF
%% final e produzido em A4 (210mm x 297mm). Isso deixa sobras enormes nas
%% margens. O pacote geometry forca margens A4 corretas, sobrepondo o
%% calculo original da classe.
\usepackage[a4paper, margin=2.5cm]{geometry}

\pagestyle{fancy}
\renewcommand{\headrulewidth}{0pt}
\fancyhead[L]{}
\fancyhead[C]{}
\fancyhead[R]{}
\fancyfoot[L]{}
\fancyfoot[C]{\small\thepage}
\fancyfoot[R]{}
\hypersetup{
  colorlinks=true,
  citecolor=blue,
  linkcolor=blue,
  urlcolor=blue,
  filecolor=blue
}

%% A classe usa \flushbottom, que estica os espacos em branco de cada pagina
%% para preenche-la ate o final. Combinado com figuras fixas ([H]), isso cria
%% buracos enormes entre uma figura/tabela e o texto seguinte. \raggedbottom
%% deixa o espaco sobrando apenas no rodape da pagina, sem esticar o meio.
\raggedbottom

%% Espacamento mais enxuto ao redor de figuras e tabelas.
\setlength{\floatsep}{8pt plus 2pt minus 2pt}
\setlength{\textfloatsep}{10pt plus 2pt minus 2pt}
\setlength{\intextsep}{8pt plus 2pt minus 2pt}

$for(header-includes)$
$header-includes$
$endfor$

  {\normalfont\rmfamily\fontseries{m}\fontsize{14}{18}\selectfont}
  {\thesubsection}{0.7em}{}
\titleformat{\subsubsection}[hang]
  {\normalfont\rmfamily\fontseries{m}\fontsize{12}{14}\selectfont}
  {\thesubsubsection}{0.7em}{}
\titlespacing*{\section}{0pt}{12pt plus 2pt minus 2pt}{6pt}

\pretocmd{\subsubsection}{\par}{}{}
\renewcommand{\contentsname}{Sumário}

%% Tipografia integralmente em Palatino: texto, titulos, legendas e tabelas.
\renewcommand{\sectionfont}{%
  \normalfont\rmfamily\fontseries{m}\fontsize{16}{18}\selectfont\color{black}}

  \normalfont\rmfamily\fontseries{m}\fontsize{14}{18}\selectfont\color{black}}

  \leftskip-\leftnoteskip\rightskip-\leftnoteskip plus1fill}

%% Referencias com indice numerico explicito, coerente com as citacoes
%% numericas empregadas ao longo do texto.
\makeatletter
\renewcommand{\@biblabel}[1]{[#1]}
\patchcmd{\thebibliography}
  {\@biblabel{\@arabic\c@enumiv.}}
  {\@biblabel{\@arabic\c@enumiv}}
  {}{}
\makeatother

%%%% Elementos pre-textuais no estilo de relatorio academico: capa formal, %%%%
%%%% pagina de resumo e pagina de sumario.                                 %%%%
%% Apos o \documentclass terminar de carregar, o LaTeX restaura o catcode de
%% @ para "outro", entao qualquer referencia a comandos internos da classe
%% (\@title, \@author, \if@subject@provided etc.) precisa ficar dentro de
%% \makeatletter ... \makeatother para ser reconhecida corretamente.
\makeatletter

%% Centraliza as legendas de figuras e tabelas (a classe original as deixa
%% alinhadas a esquerda).
\long\def\@makecaption#1#2{%
  \vskip\abovecaptionskip
  \centering{\captionheadfont #1.}\hskip2.5pt{\captionfont #2}\par
  \vskip\belowcaptionskip}

\renewenvironment{abstract}{%
  \global\setbox\absbox\vbox\bgroup\par
  \normalfont\rmfamily\fontsize{10}{14}\selectfont
  \leftskip0pc\rightskip0pc\parindent0pt\noindent\ignorespaces
}{\par\egroup}

%% Guarda apenas o conteudo dos metadados; os rotulos e a formatacao sao
%% aplicados explicitamente na pagina de resumo.
\renewcommand\subject[1]{%
  \global\@subject@providedtrue%
  \gdef\@subject{#1}}
\renewcommand\keywords[1]{%
  \global\@keywords@providedtrue%
  \gdef\@keywords{#1}}

%% A classe modeloUFCG.cls define \@maketitle atraves de mecanismos internos
%% (fila de duas colunas / bloco de revisao editorial) que absorvem qualquer
%% redefinicao de \@maketitle sem executa-la. Para evitar essa maquinaria,
%% \maketitle e redefinido diretamente aqui.
\renewcommand\maketitle{%
  \begin{titlepage}
    \thispagestyle{empty}
    \begin{center}\rmfamily
      \includegraphics[width=3.3cm]{_extensions/modeloufcg/lea_logo_bw}\par
      \vspace{0.7cm}
      $if(instituicao)$
        {\bfseries\fontsize{13}{16}\selectfont\MakeUppercase{$instituicao$}\par}
      $endif$
      $if(unidade)$
        \vspace{0.15cm}
        {\fontsize{11}{14}\selectfont $unidade$\par}
      $endif$
      $if(laboratorio)$
        {\fontsize{11}{14}\selectfont $laboratorio$\par}
      $endif$

      \vspace{1.5cm}
      {\fontsize{11}{15}\selectfont
        $for(authors)$
          $it.name.literal$$sep$\\[0.15cm]
        $endfor$\par}

      \vfill
      {\bfseries\fontsize{17}{22}\selectfont\MakeUppercase{\@title}\par}
      \vspace{1.2cm}
      $if(disciplina)$
        {\fontsize{11}{15}\selectfont
          Relatório apresentado à disciplina de \textit{$disciplina$}%
          $if(periodo)$, período $periodo$$endif$.\par}
      $endif$
      $if(professores)$
        \vspace{0.8cm}
        {\fontsize{11}{15}\selectfont
          $for(professores)$
            $professores$$sep$\\
          $endfor$\par}
      $endif$

      \vfill
      {\fontsize{11}{14}\selectfont
        $if(cidade)$$cidade$\\$endif$
        $if(ano)$$ano$$endif$\par}
    \end{center}
  \end{titlepage}

  \clearpage
  \thispagestyle{plain}
  \begin{center}
    {\rmfamily\fontseries{b}\Large RESUMO\par}
  \end{center}
  \vspace{1.2cm}
  \ifvoid\absbox\else
    \noindent\usebox\absbox\par
    \vspace{0.8cm}
  \fi
  \if@subject@provided
    {\normalfont\rmfamily\fontsize{10}{14}\selectfont
      \noindent{\bfseries Áreas temáticas: }\@subject\par}
    \vspace{0.35cm}
  \fi
  \if@keywords@provided
    {\normalfont\rmfamily\fontsize{10}{14}\selectfont
      \noindent{\bfseries Palavras-chave: }\@keywords\par}
  \fi

  \clearpage
  \thispagestyle{plain}
  \renewcommand{\contentsname}{Sumário}
  \tableofcontents

  \clearpage
  \thispagestyle{plain}
  \pagestyle{fancy}
}

\makeatother
