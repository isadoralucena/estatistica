%% Usa uma unica familia serifada em todo o relatorio. A classe carrega
%% Palatino como fonte romana; igualar \sfdefault a \rmdefault tambem cobre
%% elementos internos da classe que originalmente solicitavam Helvetica.
\renewcommand{\sfdefault}{\rmdefault}
%% Fixa cada figura exatamente onde e gerada ([H]). Sem isso, floats podem
%% "flutuar" livremente e, neste documento (muitas figuras grandes em
%% sequencia), acabam se deslocando para muito longe do texto que os
%% referencia -- inclusive interferindo na quebra de pagina de tabelas
%% longtable vizinhas.
\usepackage{float}
\floatplacement{figure}{H}

%% Evita que tabelas curtas comecem no fim da pagina e sejam interrompidas
%% por figuras na pagina seguinte. Tabelas realmente longas continuam
%% podendo quebrar normalmente entre paginas.
\BeforeBeginEnvironment{longtable}{\Needspace{12\baselineskip}}

%% A classe modeloUFCG.cls calcula largura/altura de texto e margens
%% assumindo o formato de trim de um periodico (174mm x 247mm), mas o PDF
%% final e produzido em A4 (210mm x 297mm). O geometry sobrepoe esse
%% calculo com a distribuicao usual de relatorios academicos brasileiros:
%% margem de encadernacao e superior maiores que as margens externa e
%% inferior.
\usepackage[
  a4paper,
  left=3cm,
  right=2cm,
  top=3cm,
  bottom=2cm,
  headheight=14pt,
  headsep=0.6cm,
  footskip=1cm
]{geometry}

%% Escala tipografica propria para relatorio. A classe de origem foi criada
%% para artigos de periodico e fixa o corpo em 9 pt; em pagina A4 esse corpo
%% tambem reduz excessivamente o sumario. O corpo de 11 pt com entrelinha de
%% 15 pt preserva uma boa densidade de informacao sem comprometer a leitura.
\makeatletter
\renewcommand\normalsize{%
  \@setfontsize\normalsize{11pt}{15pt}%
  \abovedisplayskip 9pt plus 3pt minus 4pt%
  \abovedisplayshortskip 0pt plus 3pt%
  \belowdisplayshortskip 6pt plus 3pt minus 3pt%
  \belowdisplayskip 9pt plus 3pt minus 4pt%
  \let\@listi\@listI}
\renewcommand\small{%
  \@setfontsize\small{10pt}{13pt}%
  \abovedisplayskip 8pt plus 3pt minus 4pt%
  \abovedisplayshortskip 0pt plus 2pt%
  \belowdisplayshortskip 5pt plus 2pt minus 2pt%
  \belowdisplayskip 8pt plus 3pt minus 4pt}
\renewcommand\footnotesize{%
  \@setfontsize\footnotesize{9pt}{11pt}%
  \abovedisplayskip 6pt plus 2pt minus 3pt%
  \abovedisplayshortskip 0pt plus 2pt%
  \belowdisplayshortskip 4pt plus 2pt minus 2pt%
  \belowdisplayskip 6pt plus 2pt minus 3pt}
\renewcommand\scriptsize{\@setfontsize\scriptsize{8pt}{9.5pt}}
\renewcommand\tiny{\@setfontsize\tiny{6pt}{7.5pt}}
\renewcommand\large{\@setfontsize\large{12.5pt}{15pt}}
\renewcommand\Large{\@setfontsize\Large{15pt}{18pt}}
\renewcommand\LARGE{\@setfontsize\LARGE{17pt}{21pt}}
\renewcommand\huge{\@setfontsize\huge{20pt}{24pt}}
\renewcommand\Huge{\@setfontsize\Huge{24pt}{29pt}}
\makeatother
\normalsize

%% O sumario recebe uma escala explicita, independente dos tamanhos herdados
%% da classe. As entradas de secao acompanham o corpo do texto; os niveis
%% seguintes diminuem gradualmente, com recuos e espacamentos proporcionais.
\makeatletter
\renewcommand{\@pnumwidth}{2.5em}
\renewcommand{\@tocrmarg}{3em}
\renewcommand*\l@section[2]{%
  \ifnum\c@tocdepth>\z@
    \addpenalty\@secpenalty
    \addvspace{0.45em \@plus 0.15em}%
    {\normalfont\rmfamily\bfseries\fontsize{11}{14}\selectfont
      \@dottedtocline{1}{0em}{2.4em}{#1}{#2}}%
  \fi}
\renewcommand*\l@subsection[2]{%
  {\normalfont\rmfamily\fontsize{10.5}{13.5}\selectfont
    \@dottedtocline{2}{2.4em}{3.2em}{#1}{#2}}}
\renewcommand*\l@subsubsection[2]{%
  {\normalfont\rmfamily\fontsize{10}{13}\selectfont
    \@dottedtocline{3}{5.6em}{3.8em}{#1}{#2}}}
\makeatother

%% A classe modeloUFCG.cls define \@maketitle atraves de mecanismos internos
%% (fila de duas colunas / bloco de revisao editorial) que absorvem qualquer
%% redefinicao de \@maketitle sem executa-la. Para evitar essa maquinaria,
%% \maketitle e redefinido diretamente aqui.
\renewcommand\maketitle{%
  \begin{titlepage}
    \thispagestyle{empty}
    \begin{center}\rmfamily
      \includegraphics[width=3.3cm]{_extensions/modeloufcg/lea_logo_bw}\par
      \vspace{0.7cm}
      $if(instituicao)$
        {\bfseries\fontsize{13}{16}\selectfont\MakeUppercase{$instituicao$}\par}
      $endif$
      $if(unidade)$
        \vspace{0.15cm}
        {\fontsize{11}{14}\selectfont $unidade$\par}
      $endif$
      $if(laboratorio)$
        {\fontsize{11}{14}\selectfont $laboratorio$\par}
      $endif$

      \vspace{1.5cm}
      {\fontsize{11}{15}\selectfont
        $for(authors)$
          $it.name.literal$$sep$\\[0.15cm]
        $endfor$\par}

      \vfill
      {\bfseries\fontsize{17}{22}\selectfont\MakeUppercase{\@title}\par}
      \vspace{1.2cm}
      $if(disciplina)$
        {\fontsize{11}{15}\selectfont
          Relatório apresentado à disciplina de \textit{$disciplina$}%
          $if(periodo)$, período $periodo$$endif$.\par}
      $endif$
      $if(professores)$
        \vspace{0.8cm}
        {\fontsize{11}{15}\selectfont
          $for(professores)$
            $professores$$sep$\\
          $endfor$\par}
      $endif$

      \vfill
      {\fontsize{11}{14}\selectfont
        $if(cidade)$$cidade$\\$endif$
        $if(ano)$$ano$$endif$\par}
    \end{center}
  \end{titlepage}

  \clearpage
  \thispagestyle{plain}
  \begin{center}
    {\rmfamily\fontseries{b}\Large RESUMO\par}
  \end{center}
  \vspace{1.2cm}
  \ifvoid\absbox\else
    \noindent\usebox\absbox\par
    \vspace{0.8cm}
  \fi
  \if@subject@provided
    {\normalfont\rmfamily\fontsize{10}{14}\selectfont
      \noindent{\bfseries Áreas temáticas: }\@subject\par}
    \vspace{0.35cm}
  \fi
  \if@keywords@provided
    {\normalfont\rmfamily\fontsize{10}{14}\selectfont
      \noindent{\bfseries Palavras-chave: }\@keywords\par}
  \fi

  \clearpage
  \thispagestyle{plain}
  \renewcommand{\contentsname}{Sumário}
  \tableofcontents

  \clearpage
  \thispagestyle{plain}
  \pagestyle{fancy}
}
